\section{User Manual}

\subsection{Getting Started}

\textbf{Mobile (Android)}
\begin{enumerate}
    \item Search ``AirWatch'' on the App Store
    \item Tap \textbf{Get}
    \item Authenticate
    \item Open the app
\end{enumerate}

\textbf{Mobile (Android)}
\begin{enumerate}
    \item Search ``AirWatch'' on Google Play
    \item Tap \textbf{Install}
    \item Open the app
\end{enumerate}

\textbf{Web}
\begin{enumerate}
    \item Navigate to \url{https://app.airwatch.io}
    \item Open in Chrome, Firefox, or Safari
    \item Tap \textbf{Install} in the browser prompt for PWA experience
\end{enumerate}

\textbf{Creating an Account}

\begin{enumerate}
    \item Tap \textbf{Get Started} on the welcome screen
    \item Enter your email address and choose a secure password (minimum 8 characters)
    \item Tap \textbf{Create Account} and verify your email
    \item Grant location permission when prompted
\end{enumerate}

\subsection{Home Screen}

The Home Screen acts as the primary dashboard.

\begin{itemize}
    \item AQI Summary Card
    \item Current Conditions
    \item 7-Day Forecast Strip
    \item Pollutant Breakdown
\end{itemize}

Data refreshes automatically every 10 minutes.

\subsection{Map Screen}

\textbf{Navigating the Map}

\begin{enumerate}
    \item Pinch to zoom and drag to pan
    \item Tap colored zones to view AQI readings
    \item AQI heat-map uses EPA color standards
\end{enumerate}

\textbf{Searching for a Location}

\begin{enumerate}
    \item Tap the search bar
    \item Type a city name
    \item Select a suggestion
    \item Save the location using the bookmark icon
\end{enumerate}

\textbf{Saved Locations}

\begin{itemize}
    \item Displays AQI badges and weather summaries
    \item Swipe left to delete
    \item Tap and hold to reorder
\end{itemize}

\subsection{Alerts \& Notifications}

\textbf{Enabling Notifications}

\begin{enumerate}
    \item Open Settings $\rightarrow$ Notifications
    \item Toggle \textbf{Enable Alerts}
    \item Grant notification permissions
\end{enumerate}

\textbf{Configuring Alert Thresholds}

\begin{enumerate}
    \item Open Settings $\rightarrow$ Notifications $\rightarrow$ Manage Locations
    \item Adjust AQI threshold sliders
    \item Receive push notifications when AQI exceeds threshold
\end{enumerate}

\textbf{Quiet Hours}

\begin{enumerate}
    \item Open Settings $\rightarrow$ Notifications $\rightarrow$ Quiet Hours
    \item Enable Quiet Hours
    \item Configure start and end times
\end{enumerate}

\subsection{Health Advisories}

Each location detail screen includes:

\begin{itemize}
    \item Category labels and badges
    \item Risk groups
    \item Recommended actions
    \item Outdoor activity guidance
\end{itemize}

\section{Design Specification}

\subsection{Design Principles}

\begin{enumerate}
    \item \textbf{Clarity First} -- pollution and health data must be instantly scannable
    \item \textbf{Color as Communication} -- AQI colors follow EPA standards
    \item \textbf{Accessibility by Default} -- minimum 4.5:1 contrast ratio
    \item \textbf{Platform Consistency} -- follows iOS and Material Design conventions
\end{enumerate}
